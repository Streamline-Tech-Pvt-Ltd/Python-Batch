*Constraints :

 1. NOT NULL:
In NOT NULL Column cannot store NULL value
ex.
CREATE TABLE student2(sr_no INT, name VARCHAR(20) NOT NULL);

2. UNIQUE:
IN UNIQUE No duplicate values allowed
ex.
CREATE TABLE student2(sr_no INT, roll_no INT UNIQUE);

3. PRIMARY KEY:
In PRIMARY KEY Uniquely identifies each row Cannot be NULL Only one per table
ex.
CREATE TABLE student3(sr_no INT PRIMARY KEY, roll_no INT UNIQUE);

4. AUTO_INCREMENT:
Automatically increases value
ex.
CREATE TABLE student4(sr_no INT AUTO_INCREMENT PRIMARY KEY, roll_no INT UNIQUE);
INSERT INTO student4(roll_no) VALUES(10);

 5. DEFAULT:
It will be Assigns default value if not provided
ex.
CREATE TABLE student5(sr_no INT, admission_year INT DEFAULT '2022');

6. CHECK:
It will be Restricts the values based on condition
ex.
CREATE TABLE student6(sr_no INT,age INT CHECK(age > 18));

7. FOREIGN KEY:
It Maintains the relationship between tables and Ensures referential integrity
ex.
Parent Table:
CREATE TABLE department(dept_id INT PRIMARY KEY, d_name VARCHAR(20));
insert into department values(1,"Java Developer");
insert into department values(2,"Python Developer");
insert into department values(3,"Data Analytics");

Child Table:
CREATE TABLE employees(emp_id INT, e_name varchar(20), E_role VARCHAR(20), salary FLOAT, dept_id INT, FOREIGN KEY (dept_id) REFERENCES department(dept_id));
INSERT INTO employees VALUES(101,"rushi","Java Developer",50000,1);
INSERT INTO employees VALUES(102,"Pramod", "Python Developer",70000,2);
